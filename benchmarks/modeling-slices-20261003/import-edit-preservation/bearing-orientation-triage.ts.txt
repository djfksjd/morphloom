import {writeFileSync} from 'node:fs';import * as THREE from 'three';
import {generateBearingProject} from '../src/engine/bearing-pack';import {buildElementScene} from '../src/engine/element-renderer';import {analyzeTopology} from '../src/engine/topology';
const project=generateBearingProject({}),built=buildElementScene(project,'detail');const rows=[];built.root.updateMatrixWorld(true);
for(const child of built.root.getObjectsByProperty('type','Mesh')){const mesh=child as THREE.Mesh,g=mesh.geometry,pos=g.getAttribute('position'),index=g.index,center=new THREE.Box3().setFromObject(mesh).getCenter(new THREE.Vector3());let volume=0;const a=new THREE.Vector3(),b=new THREE.Vector3(),c=new THREE.Vector3();for(let i=0;i<(index?.count??pos.count);i+=3){a.fromBufferAttribute(pos,index?index.getX(i):i).applyMatrix4(mesh.matrixWorld).sub(center);b.fromBufferAttribute(pos,index?index.getX(i+1):i+1).applyMatrix4(mesh.matrixWorld).sub(center);c.fromBufferAttribute(pos,index?index.getX(i+2):i+2).applyMatrix4(mesh.matrixWorld).sub(center);volume+=a.dot(b.cross(c))/6;}rows.push({id:mesh.name,signedVolumeMm3:volume*1e9,topology:analyzeTopology(mesh),triangles:(index?.count??pos.count)/3});}
writeFileSync(process.argv[2]+'/before.project.json',JSON.stringify(project,null,2));writeFileSync(process.argv[2]+'/before-orientation.json',JSON.stringify({renderer:'morphloom.element-renderer/0.10',rows},null,2));console.log(JSON.stringify(rows.map(r=>({id:r.id,signedVolumeMm3:r.signedVolumeMm3,localWinding:r.topology.inconsistentWindingEdges,closed:r.topology.watertightMeshes})),null,2));built.dispose();
